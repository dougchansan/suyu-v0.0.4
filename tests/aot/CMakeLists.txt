cmake_minimum_required(VERSION 3.20)
project(suyu_aot_contract_tests LANGUAGES CXX)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
option(AOT_TEST_SANITIZERS "Enable address/undefined-behavior sanitizers" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(SUYU_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/make_support.py"
                        "${CMAKE_CURRENT_BINARY_DIR}/support"
                RESULT_VARIABLE SUPPORT_RESULT)
if(NOT SUPPORT_RESULT EQUAL 0)
    message(FATAL_ERROR "Cannot create the explicit component-test dependency doubles")
endif()
add_executable(aot_metrics_tests test_metrics.cpp)
add_executable(aot_perf_stats_tests test_perf_stats.cpp "${SUYU_ROOT}/src/core/perf_stats.cpp")
target_include_directories(aot_perf_stats_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/support")
target_link_libraries(aot_perf_stats_tests PRIVATE Threads::Threads)
foreach(TARGET aot_metrics_tests aot_perf_stats_tests)
    target_include_directories(${TARGET} PRIVATE "${SUYU_ROOT}/src")
    if(MSVC)
        target_compile_options(${TARGET} PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(${TARGET} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        if(AOT_TEST_SANITIZERS)
            target_compile_options(${TARGET} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${TARGET} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endforeach()
add_test(NAME aot_metrics COMMAND aot_metrics_tests "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME aot_perf_stats_component COMMAND aot_perf_stats_tests "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME aot_trace_comparison COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_compare.py"
                                         "${CMAKE_CURRENT_BINARY_DIR}")
set_tests_properties(aot_trace_comparison PROPERTIES DEPENDS "aot_metrics;aot_perf_stats_component")
