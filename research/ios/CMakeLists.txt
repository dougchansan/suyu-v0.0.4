cmake_minimum_required(VERSION 3.24)
project(iHorizon LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SUYU_NO_JIT "Research lane must not contain a JIT" ON)
if(NOT SUYU_NO_JIT)
    message(FATAL_ERROR "This research target requires SUYU_NO_JIT=ON")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "64-bit target required")
endif()
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
set(SWITCH_AOT_EXEFS "" CACHE PATH "Private export outside Git; empty uses synthetic fixture")
option(SWITCH_AOT_TEST_OMIT_SYMBOL_FIX "Negative link test only" OFF)
if(NOT SWITCH_AOT_EXEFS)
    set(SWITCH_AOT_EXEFS "${CMAKE_CURRENT_BINARY_DIR}/synthetic-exefs")
    execute_process(COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/make_fixture.py" "${SWITCH_AOT_EXEFS}"
        COMMAND_ERROR_IS_FATAL ANY)
endif()
set(is_synthetic OFF)
if(EXISTS "${SWITCH_AOT_EXEFS}/SYNTHETIC_FIXTURE.txt")
    set(is_synthetic ON)
endif()
if(SWITCH_AOT_TEST_OMIT_SYMBOL_FIX AND NOT is_synthetic)
    message(FATAL_ERROR "Negative tests require synthetic inputs")
endif()
include(cmake/StaticModules.cmake)
switch_aot_add_images(switch_aot_images "${SWITCH_AOT_EXEFS}")
add_library(switch_aot_registry STATIC src/static_registry.cpp)
target_include_directories(switch_aot_registry PUBLIC include)
target_compile_definitions(switch_aot_registry PUBLIC SUYU_NO_JIT=1)
add_library(switch_aot_host STATIC src/research_host.cpp)
target_link_libraries(switch_aot_host PUBLIC switch_aot_registry switch_aot_images)
if(is_synthetic)
    target_compile_definitions(switch_aot_host PRIVATE SWITCH_AOT_SYNTHETIC=1)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    enable_language(OBJCXX)
    # iHorizon has a stable identity, separate from other applications.
    set(SWITCH_AOT_BUNDLE_ID "org.ihorizon.app" CACHE STRING "iHorizon app identity")
    add_executable(iHorizon MACOSX_BUNDLE apple/main.mm)
    target_compile_options(iHorizon PRIVATE -fobjc-arc)
    target_link_libraries(iHorizon PRIVATE switch_aot_host "-framework UIKit")
    set_target_properties(iHorizon PROPERTIES
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/apple/Info.plist.in"
        MACOSX_BUNDLE_GUI_IDENTIFIER "${SWITCH_AOT_BUNDLE_ID}"
        XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "${SWITCH_AOT_BUNDLE_ID}"
        XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2"
        XCODE_ATTRIBUTE_CODE_SIGN_STYLE Automatic
        XCODE_ATTRIBUTE_CODE_SIGN_ENTITLEMENTS "${CMAKE_CURRENT_SOURCE_DIR}/apple/Research.entitlements"
        XCODE_GENERATE_SCHEME TRUE)
else()
    add_executable(switch_aot_probe src/probe_main.cpp)
    target_link_libraries(switch_aot_probe PRIVATE switch_aot_host)
    enable_testing()
    add_test(NAME static_module_probe COMMAND switch_aot_probe)
endif()
